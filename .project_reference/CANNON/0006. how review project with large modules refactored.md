Phase 2 — AST Name Resolution Engine

This is the core. For each helper module, we build four sets then diff them:

USED     = all Name/Attribute loads in the AST
DEFINED  = local assignments + imports + def/class + parameters + builtins
EXPORTED = symbols from root facade + sibling helpers (read their __all__ or top-level defs)

UNRESOLVED = USED - DEFINED - EXPORTED - BUILTINS

Key AST visitor logic:
python

import ast, builtins

BUILTIN_NAMES = set(dir(builtins))

class NameAuditVisitor(ast.NodeVisitor):
    def __init__(self):
        self.loaded   = set()  # Name nodes with Load context
        self.stored   = set()  # Name nodes with Store context
        self.imported = set()  # from import / import as
        self.defined  = set()  # def / class / lambda args
        self.augmented = set() # AugAssign targets

    def visit_Name(self, node):
        if isinstance(node.ctx, ast.Load):
            self.loaded.add(node.id)
        elif isinstance(node.ctx, ast.Store):
            self.stored.add(node.id)
        self.generic_visit(node)

    def visit_Import(self, node):
        for alias in node.names:
            name = alias.asname or alias.name.split(".")[0]
            self.imported.add(name)
        self.generic_visit(node)

    def visit_ImportFrom(self, node):
        for alias in node.names:
            self.imported.add(alias.asname or alias.name)
        self.generic_visit(node)

    def visit_FunctionDef(self, node):
        self.defined.add(node.name)
        for arg in node.args.args:
            self.defined.add(arg.arg)
        self.generic_visit(node)

    visit_AsyncFunctionDef = visit_FunctionDef

    def visit_ClassDef(self, node):
        self.defined.add(node.name)
        self.generic_visit(node)

import subprocess, sys

def import_smoke(module_path, project_root):
    result = subprocess.run(
        [sys.executable, "-c",
         f"import importlib.util; "
         f"spec = importlib.util.spec_from_file_location('_smoke', r'{module_path}'); "
         f"mod = importlib.util.module_from_spec(spec); "
         f"spec.loader.exec_module(mod)"],
        cwd=project_root,
        env={**os.environ, "PYTHONPATH": project_root},
        capture_output=True, text=True, timeout=15
    )
    return result.returncode, result.stderr

Structural Safeguards

Three rules the runner must enforce about itself:

    Never exec() or eval() project source — use ast.parse() only for static analysis, subprocess for execution smoke
    Temp files in system temp dir, never inside E:\developer_tools
    Exit code 0 always — the audit runner itself must not crash; catch all per-module exceptions and log them as findings
