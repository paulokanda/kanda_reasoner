def read_text_utf8(path: Path) -> str:
    return path.read_text(encoding="utf-8", errors="replace")


def write_text_utf8(path: Path, text: str) -> None:
    path.write_text(text, encoding="utf-8", newline="\n")


def read_json(path: Path) -> Any:
    return json.loads(read_text_utf8(path))


def write_json(path: Path, data: Any) -> None:
    text = json.dumps(data, indent=2, ensure_ascii=False)

def detect_workspace_root(script_path: Path, explicit_workspace: Path | None) -> Path:
    if explicit_workspace is not None:
        return explicit_workspace.resolve()

    candidates = [
        script_path.parent.parent,
        Path.cwd(),
    ]

    for candidate in candidates:
        if (candidate / PROMPT_LIBRARY_DIR_NAME).exists():
            return candidate.resolve()

    return script_path.parent.parent.resolve()


def load_source_map(source_map_path: Path) -> dict[str, Any]:
    if not source_map_path.exists():
        raise FileNotFoundError(f"Missing source map: {source_map_path}")

    raw = read_json(source_map_path)
    if not isinstance(raw, dict):
        raise ValueError("Source map must be a JSON object.")

    sources = raw.get("startup_sources")
    if not isinstance(sources, list):
        raise ValueError("Source map must contain a startup_sources list.")

    return raw


def validate_source_map(workspace_root: Path, source_map: dict[str, Any]) -> list[str]:
    warnings: list[str] = []
    sources = source_map["startup_sources"]

    seen_orders: set[int] = set()
    seen_generated: set[str] = set()
    seen_prompt_ids: set[str] = set()

    for item in sources:
        load_order = int(item["load_order"])
        canonical_source = str(item["canonical_source"])
        generated_filename = str(item["generated_filename"])
        prompt_id = str(item["prompt_id"])

        if load_order in seen_orders:
            warnings.append(f"Duplicate load_order: {load_order}")
        seen_orders.add(load_order)

        if generated_filename in seen_generated:
            warnings.append(f"Duplicate generated_filename: {generated_filename}")
        seen_generated.add(generated_filename)

        if prompt_id in seen_prompt_ids:
            warnings.append(f"Duplicate prompt_id: {prompt_id}")
        seen_prompt_ids.add(prompt_id)

        if "\\" in canonical_source:
            warnings.append(f"Use forward slashes in canonical_source: {canonical_source}")

        source_path = workspace_root / canonical_source
        if not source_path.exists():
            warnings.append(f"Mapped source does not exist: {canonical_source}")

    expected_orders = list(range(1, len(sources) + 1))
    actual_orders = sorted(int(item["load_order"]) for item in sources)
    if actual_orders != expected_orders:
        warnings.append(
            "load_order values should be contiguous. "
            f"Expected {expected_orders}, got {actual_orders}"
        )

    return warnings


def prompt_id_from_filename(path: Path) -> str:
    name = path.stem.lower()
    cleaned = []
    for char in name:
        if char.isalnum():
            cleaned.append(char)
        else:
            cleaned.append("_")
    prompt_id = "".join(cleaned).strip("_")
    while "__" in prompt_id:
        prompt_id = prompt_id.replace("__", "_")
    return prompt_id


def parse_bool_marker(text: str, key: str) -> bool:
    target = key.lower()
    for line in text.splitlines():
        stripped = line.strip()
        if not stripped or ":" not in stripped:
            continue
        left, right = stripped.split(":", 1)
        if left.strip().lower() == target:
            return right.strip().lower() in {"true", "yes", "1"}
    return False


def parse_text_marker(text: str, key: str) -> str | None:
    target = key.lower()
    for line in text.splitlines():
        stripped = line.strip()
        if not stripped or ":" not in stripped:
            continue
        left, right = stripped.split(":", 1)
        if left.strip().lower() == target:
            value = right.strip()
            return value if value else None
    return None


def candidate_from_meta(
    workspace_root: Path,
    md_path: Path,
    meta_path: Path,
) -> StartupCandidate | None:
    data = read_json(meta_path)
    if not isinstance(data, dict):
        return None

    if data.get("startup_kernel_include") is not True:
        return None

    canonical_source = str(md_path.relative_to(workspace_root)).replace("\\", "/")
    generated_filename = str(
        data.get("startup_generated_filename")
        or md_path.name
    )

    prompt_id = str(
        data.get("prompt_id")
        or data.get("startup_prompt_id")
        or prompt_id_from_filename(md_path)
    )

    load_mode = str(data.get("startup_load_mode") or "always_startup")
    role = str(data.get("startup_role") or "Startup prompt request kernel file.")

    return StartupCandidate(
        canonical_source=canonical_source,
        generated_filename=generated_filename,
        prompt_id=prompt_id,
        load_mode=load_mode,
        role=role,
    )


def candidate_from_markers(
    workspace_root: Path,
    md_path: Path,
) -> StartupCandidate | None:
    text = read_text_utf8(md_path)

    if not parse_bool_marker(text, "Startup kernel include"):
        return None

    canonical_source = str(md_path.relative_to(workspace_root)).replace("\\", "/")

    generated_filename = (
        parse_text_marker(text, "Startup generated filename")
        or md_path.name
    )

    prompt_id = (
        parse_text_marker(text, "Startup prompt id")
        or prompt_id_from_filename(md_path)
    )

    load_mode = (
        parse_text_marker(text, "Startup load mode")
        or "always_startup"
    )

    role = (
        parse_text_marker(text, "Startup role")
        or "Startup prompt request kernel file."
    )

    return StartupCandidate(
        canonical_source=canonical_source,
        generated_filename=generated_filename,
        prompt_id=prompt_id,
        load_mode=load_mode,
        role=role,
    )


def find_startup_candidates(workspace_root: Path) -> list[StartupCandidate]:
    prompt_library = workspace_root / PROMPT_LIBRARY_DIR_NAME
    if not prompt_library.exists():
        raise FileNotFoundError(f"Missing prompt_library: {prompt_library}")

    candidates: list[StartupCandidate] = []

    search_roots = [
        prompt_library / "ACTIVE_PROMPTS",
        prompt_library / "ROUTING",
    ]

    for search_root in search_roots:
        if not search_root.exists():
            continue

        for md_path in sorted(search_root.rglob("*.md")):
            meta_path = md_path.with_suffix(".meta.json")

            candidate = None
            if meta_path.exists():
                candidate = candidate_from_meta(workspace_root, md_path, meta_path)

            if candidate is None:
                candidate = candidate_from_markers(workspace_root, md_path)

            if candidate is not None:
                candidates.append(candidate)

    return candidates


def current_source_keys(source_map: dict[str, Any]) -> set[str]:
    keys: set[str] = set()
    for item in source_map["startup_sources"]:
        keys.add(str(item["canonical_source"]).replace("\\", "/"))
    return keys


def find_missing_candidates(
    source_map: dict[str, Any],
    candidates: list[StartupCandidate],
) -> list[StartupCandidate]:
    existing = current_source_keys(source_map)
    return [
        candidate
        for candidate in candidates
        if candidate.canonical_source not in existing
    ]

def apply_missing_candidates(
    source_map: dict[str, Any],
    missing: list[StartupCandidate],
) -> dict[str, Any]:
    updated = dict(source_map)
    sources = list(updated["startup_sources"])

    next_order = max(int(item["load_order"]) for item in sources) + 1 if sources else 1

    for candidate in missing:
        sources.append(
            {
                "load_order": next_order,
                "canonical_source": candidate.canonical_source,
                "generated_filename": candidate.generated_filename,
                "prompt_id": candidate.prompt_id,
                "load_mode": candidate.load_mode,
                "role": candidate.role,
            }
        )
        next_order += 1

    updated["startup_sources"] = sources
    return updated

Overall Verdict: Keep with modifications

The core idea is correct. The governed, opt-in, report-first design philosophy is right. Several specific decisions need adjustment.
Recommended Architecture

The proposed component list is good but needs two additions and one split:

    Add a validate_candidate_fields(candidate) function that runs before either reporting or applying — not just at apply time.
    Split find_startup_candidates() into find_candidates_from_meta() and find_candidates_from_markers() explicitly (the proposal names them but conflates them in the main loop description). They have different trust levels and should be processed separately with different warning levels.
    Add a detect_stale_source_map_entries() function (read-only, report only) even in v1. Not having it means the source map silently accumulates ghost entries when files are deleted or renamed.

Minimal v1 Behavior

Include only:

    --check (default): scan, compare, produce STARTUP_ROUTING_KERNEL_SOURCES_UPDATE_CANDIDATE.md, print STATUS.
    --apply --yes: modify source map, validate before and after, never touch first_AI_deliver/.
    --apply --sync --yes: same as above, then invoke existing generator.
    Stale entry detection: report only, no removal in v1.
    Rename detection: omit entirely from v1.

Do not include in v1: automatic removal, rename tracking, CI integration, non-numbered filename support.
Marker vs Sidecar — Recommendation

Support both, but make sidecar the authoritative source. The practical reason: Markdown markers are fragile — a prompt author can accidentally break the marker format by editing the header, and the scanner will silently miss the file. A sidecar .meta.json is a distinct artifact that is harder to corrupt accidentally.

Concrete policy:

    If both exist for the same file, sidecar wins. Log a warning that both exist.
    Markdown markers are accepted as a fallback but emit a WARNING: Markdown marker detected — consider migrating to sidecar in the candidate report.
    This gives you a natural migration path: authors start with markers, graduate to sidecars.

Mandatory Fields for a Startup Candidate

These must all be present and non-empty, or the candidate is rejected with a clear per-field error:
Field	Reason
startup_kernel_include: true	Explicit opt-in required
startup_load_mode	Must be always_startup for v1; future modes reserved
startup_generated_filename	Must be unique across all source-map entries
prompt_id	Must be unique, alphanumeric + underscores only
startup_role	Human-readable; required for the candidate report to be useful
canonical_source	Relative forward-slash path; must exist at scan time

Optional but warned-if-absent: load_order. If absent, tool assigns next available integer and flags it in the report for human confirmation.
Apply Policy

Yes, --apply should exist, but with a hard double-gate: --apply --yes together, not --apply alone. Your proposal already has this right. One addition: before writing, the tool should serialize the proposed new source map to a temp location, validate it with validate_source_map(), and only overwrite the real file if validation passes. If validation fails after applying, it must restore the original and exit non-zero. This is an atomic write pattern — critical for a governed source map.
Safety Checks to Add

The proposal misses these:

    Encoding check: Reject any candidate file that cannot be read as UTF-8. Silent encoding errors in prompt files cause subtle downstream corruption.
    File size guard: Warn if a candidate file exceeds a threshold (suggest 100KB). Oversized prompts being added to the daily startup ZIP is a real operational risk.
    Duplicate canonical_source check: A file could have two sidecars pointing to it, or the source map could already list it under a different prompt_id. Detect this.
    load_order gap detection: Your proposal says "ensure load_order remains contiguous" but doesn't say what to do if the human assigns load_order: 15 when the current max is 11. The tool should warn about gaps and refuse to apply if they exist, rather than silently accepting them.
    Source map schema version field: Add a schema_version key to STARTUP_ROUTING_KERNEL_SOURCES.json now, even if v1 only supports version 1. This prevents future migration pain.
    No-op apply detection: If --apply --yes is run and there is nothing to apply (STATUS: NO_UPDATE_NEEDED), the tool should exit cleanly with a clear message rather than writing an identical file or erroring.
    Workspace root guard: detect_workspace_root() should verify the expected directory structure exists before proceeding. If it can't confirm it's inside a valid kanda workspace, it should refuse to run. This prevents accidental execution from a wrong directory.

Stale Entry and Rename Handling

Stale entries: Detect in v1, report only. The report should list any source-map entry whose canonical_source file no longer exists. Do not remove automatically in v1. Add --remove-stale --yes as a v2 feature only.

Rename detection: Omit from v1 entirely. Rename detection via hash comparison against the source map is tempting but introduces significant risk — a prompt that was edited and moved looks identical to a rename and a fresh edit together. The failure mode (silently treating a new file as a renamed old one) is worse than the manual effort of updating the source map by hand.
Generated Filename Numbering

Yes, validate that the number is the next available sequence. But: do not auto-assign silently. The tool should propose the next number in the candidate report and require the human to confirm it in the sidecar before --apply will proceed. This keeps load order human-governed while catching mistakes.

Non-numbered filenames: Do not allow in v1. The numbering is part of the load-order contract. If you allow unnumbered names, you lose the ability to reason about load order from filenames alone. Reserve this for a future explicit decision.
Sync Integration

--sync invoking the existing generator after --apply is the right design. One implementation note: invoke the generator as a subprocess with subprocess.run([sys.executable, generator_path, "--ensure-sync", "--yes"]) and capture its exit code. If the generator exits non-zero, the tool should report SYNC FAILED and exit non-zero itself — do not swallow sync failures.
Failure Modes to Test

These are the ones most likely to cause silent damage:

    Sidecar exists but canonical_source path points to a deleted file — should report stale, not crash.
    Two sidecars with the same prompt_id — must be caught before apply.
    --apply --yes run when source map is currently malformed — must refuse to apply, not overwrite.
    --apply --sync --yes where sync generator fails — source map update already applied; tool must clearly report partial state.
    Markdown marker present but sidecar also present with conflicting fields — sidecar wins, warning emitted.

    Tool run from wrong directory (not workspace root) — workspace root guard must catch this.
    Candidate file is not UTF-8 — must reject cleanly.
    Source map JSON is syntactically invalid — must report clearly, not traceback.
    --apply without --yes — must print usage and refuse.

Risk	Mitigation
Author adds marker to a draft prompt, accidentally gets it into startup ZIP	File must be in an approved scan folder AND have valid sidecar. Keep draft prompts in a folder outside the scan scope.

Tool run by AI agent without human review	The --yes gate exists, but document explicitly that this tool should never be invoked autonomously by an AI without human confirmation of the candidate report.
Specific Corrections to the Proposed Implementation

    The proposed main() flow applies before validating candidates. Reverse this: validate all candidates first, fail early if any are invalid, then report or apply.
    The proposal doesn't specify what happens if STARTUP_ROUTING_KERNEL_SOURCES_UPDATE_CANDIDATE.md already exists from a previous run. It should be overwritten with a timestamp header, not appended to.

    The --sync flag description says it "optionally runs" the generator. Make explicit in both docs and argparse help that --sync requires --apply — --sync alone should be rejected with a usage error.
    Plain text output requirement is good, but add a machine-readable summary line at the end of stdout (the STATUS: lines) so the output can be parsed by a calling script without screen-scraping prose.

Final Recommendation

Build v1 with check, apply, and sync as proposed, with the corrections above. Keep rename detection and auto-removal firmly in v2. The sidecar-wins-over-marker policy resolves the dual-format ambiguity cleanly and gives you a migration path without breaking existing workflows.

        finally {
